sudo ip link set down ens3
sudo ip link set down ens4
sudo ip link add bond0 type bond mode 802.3ad 
sudo ip link set dev bond0 type bond lacp_active on
sudo ip link set dev bond0 type bond lacp_rate fast
sudo ip link set ens3 master bond0
sudo ip link set ens4 master bond0
sudo ip link set up ens3
sudo ip link set up ens4
sudo ip link add link bond0 name v210.bond0 type vlan id 210
sudo ip addr add 192.168.210.1/24 dev v210.bond0
sudo ip link set up bond0
sudo ip link set up v210.bond0
sudo ip r add default via 192.168.210.254
